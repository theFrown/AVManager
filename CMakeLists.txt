cmake_minimum_required(VERSION 3.25)
project(AVManager LANGUAGES CXX)
add_library(avmanager_core STATIC src/avr_handler.cpp src/keyboard_hook.cpp src/logger.cpp)
add_executable(AVManager src/main.cpp)
target_compile_features(AVManager PRIVATE cxx_std_20)
target_compile_features(avmanager_core PUBLIC cxx_std_20)
target_link_libraries(AVManager PRIVATE avmanager_core)
target_link_libraries(avmanager_core PRIVATE ws2_32)
target_compile_definitions(avmanager_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_include_directories(avmanager_core PUBLIC src)
if(MSVC)
    target_compile_options(AVManager PRIVATE /W4)
    target_compile_options(avmanager_core PRIVATE /W4)
endif()
