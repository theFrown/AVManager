cmake_minimum_required(VERSION 3.20)
project(AVManager)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(AVManager main.cpp AVRHandler.cpp keyboard.cpp)
target_link_libraries(AVManager ws2_32)
target_compile_definitions(AVManager PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
if(MSVC)
    target_compile_options(AVManager PRIVATE /W4)
endif()
